cmake_minimum_required(VERSION 3.20)
project(resurge_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(resurge_engine src/solver.cpp)
target_include_directories(resurge_engine PUBLIC include)
target_compile_options(resurge_engine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(resurge-engine src/main.cpp)
target_link_libraries(resurge-engine PRIVATE resurge_engine)
add_executable(resurge-engine-tests tests/test_solver.cpp)
target_link_libraries(resurge-engine-tests PRIVATE resurge_engine)
enable_testing()
add_test(NAME solver COMMAND resurge-engine-tests)
